### Database Setup

```sql
CREATE TABLE `courses` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `name` VARCHAR(255) NOT NULL,
  `thumbnail` VARCHAR(255), 
  `description`TEXT,
  `status` VARCHAR(50) DEFAULT 'active'
);

CREATE TABLE `sub_courses` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `course_id` INT,
  `name` VARCHAR(255) NOT NULL,
  FOREIGN KEY (`course_id`) REFERENCES `courses`(`id`)
);

CREATE TABLE `quizzes` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `sub_course_id` INT,
  `question` TEXT,
  `option_a` TEXT,
  `option_b` TEXT,
  `option_c` TEXT,
  `option_d` TEXT,
  `correct_option` VARCHAR(1),
  `points` INT,
  FOREIGN KEY (`sub_course_id`) REFERENCES `sub_courses`(`id`)
);

CREATE TABLE `users` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `username` VARCHAR(255) NOT NULL,
  `score` INT DEFAULT 0
);

CREATE TABLE `user_answers` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `user_id` INT,
  `quiz_id` INT,
  `selected_option` VARCHAR(1),
  `correct` BOOLEAN,
  `points_earned` INT,
  FOREIGN KEY (`user_id`) REFERENCES `users`(`id`),
  FOREIGN KEY (`quiz_id`) REFERENCES `quizzes`(`id`)
);

CREATE TABLE `user_courses` (
  `user_id` INT,
  `sub_course_id` INT,
  `completed` BOOLEAN DEFAULT FALSE,
  PRIMARY KEY (`user_id`, `sub_course_id`),
  FOREIGN KEY (`user_id`) REFERENCES `users`(`id`),
  FOREIGN KEY (`sub_course_id`) REFERENCES `sub_courses`(`id`)
);
```

### Update User Courses Page (user_courses.php) to Disable Completed Courses

Ensure that completed courses are disabled on the selection page:

```php
<?php
session_start();
$conn = new mysqli('localhost', 'username', 'password', 'database');
$user_id = 1; // For demonstration purposes, assuming user_id = 1.

$courses_result = $conn->query("SELECT * FROM courses");
$result = $conn->query("SELECT sc.id, sc.name, uc.completed FROM sub_courses sc LEFT JOIN user_courses uc ON sc.id = uc.sub_course_id AND uc.user_id = $user_id");

?>

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>User Courses</title>
    <link href="https://cdn.jsdelivr.net/npm/tailwindcss@2.2.19/dist/tailwind.min.css" rel="stylesheet">
</head>
<body>
<div class="container mx-auto p-6">
    <h1 class="text-3xl font-bold mb-4">Select a Course</h1>
    <form method="post" class="space-y-4">
        <div>
            <label for="sub_course_id">Sub-Course</label>
            <select name="sub_course_id" id="sub_course_id" class="block w-full mt-1">
                <?php while ($sub_course = $result->fetch_assoc()): ?>
                    <option value="<?= $sub_course['id'] ?>" <?= $sub_course['completed'] ? 'disabled' : '' ?>>
                        <?= $sub_course['name'] ?>
                    </option>
                <?php endwhile; ?>
            </select>
        </div>
        <button type="submit" class="bg-blue-500 text-white px-4 py-2 rounded mt-4">Start Quiz</button>
        <h1 class="text-3xl font-bold mb-4">Available Courses</h1>
        <div class="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
            <?php while ($course = $courses_result->fetch_assoc()): ?>
                <div class="bg-white p-4 rounded shadow">
                    <img src="<?= $course['thumbnail'] ?>" alt="<?= $course['name'] ?>" class="w-full h-48 object-cover rounded">
                    <h2 class="text-xl font-bold mt-4"><?= $course['name'] ?></h2>
                    <p class="mt-2"><?= $course['description'] ?></p>
                    <form action="user_subcourses.php" method="GET" class="mt-4">
                        <input type="hidden" name="course_id" value="<?= $course['id'] ?>">
                        <button type="submit" class="bg-blue-500 text-white px-4 py-2 rounded">Start Learning</button>
                    </form>
                </div>
            <?php endwhile; ?>
        </div>
        
    </form>
</div>
</body>
</html>
```

Now, when the user completes a quiz, the corresponding sub-course will be marked as completed and disabled in the course selection page.
